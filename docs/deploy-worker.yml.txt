name: Deploy the pick'em contest API

# The contest backend (worker/) — the one server in this repo. Applies any new D1 migrations and
# deploys the Cloudflare Worker whenever worker/ changes. Needs two repo secrets:
# CLOUDFLARE_API_TOKEN (Workers Scripts: Edit + D1: Edit) and CLOUDFLARE_ACCOUNT_ID.
# This is NOT a data cron — the tracker still reads ESPN live in the browser; the Worker only
# keeps the contest ledger (entries, picks, finished weeks).

on:
  push:
    branches: [main]
    paths: ['worker/**']
  workflow_dispatch:

permissions:
  contents: read

jobs:
  deploy:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: worker
    steps:
      - uses: actions/checkout@v4
      # Until WPR adds the Cloudflare secrets (worker/README.md), there is nothing to deploy to:
      # say so and go green rather than fail on every push that touches worker/.
      - name: Check for Cloudflare credentials
        id: creds
        env:
          TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          ACCOUNT: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
        run: |
          if [ -n "$TOKEN" ] && [ -n "$ACCOUNT" ]; then
            echo "configured=true" >> "$GITHUB_OUTPUT"
          else
            echo "configured=false" >> "$GITHUB_OUTPUT"
            echo "::notice::Contest API not deployed — add the CLOUDFLARE_API_TOKEN and CLOUDFLARE_ACCOUNT_ID repo secrets (see worker/README.md)."
          fi
      - uses: actions/setup-node@v4
        if: steps.creds.outputs.configured == 'true'
        with:
          node-version: 20
      - if: steps.creds.outputs.configured == 'true'
        run: npm install --no-audit --no-fund
      - name: Apply D1 migrations
        if: steps.creds.outputs.configured == 'true'
        run: npx wrangler d1 migrations apply wpr-packers-pickem --remote
        env:
          CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
      - name: Deploy the Worker
        if: steps.creds.outputs.configured == 'true'
        run: npx wrangler deploy
        env:
          CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          CLOUDFLARE_ACCOUNT_ID: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
